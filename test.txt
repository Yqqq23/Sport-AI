#!/usr/bin/env bash

set -euo pipefail

REPO_ROOT="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
VENV_DIR="${REPO_ROOT}/.venv"
VIDEO_DIR="${REPO_ROOT}/videos"

TRACKER="${TRACKER:-bytetrack.yaml}"

if [ ! -f "${VENV_DIR}/bin/activate" ]; then
    echo "ERROR: virtual environment not found: ${VENV_DIR}"
    echo "Run: bash install_server.sh"
    exit 1
fi

if [ "$#" -ge 1 ]; then
    if [ -f "$1" ]; then
        VIDEO_PATH="$(cd "$(dirname "$1")" && pwd)/$(basename "$1")"
    else
        if [ ! -d "${VIDEO_DIR}" ]; then
            echo "ERROR: video directory not found: ${VIDEO_DIR}"
            exit 1
        fi
        VIDEO_PATH="${VIDEO_DIR}/$1"
    fi
else
    if [ ! -d "${VIDEO_DIR}" ]; then
        echo "ERROR: video directory not found: ${VIDEO_DIR}"
        exit 1
    fi
    shopt -s nullglob nocaseglob
    VIDEOS=(
        "${VIDEO_DIR}"/*.mp4
        "${VIDEO_DIR}"/*.avi
        "${VIDEO_DIR}"/*.mov
        "${VIDEO_DIR}"/*.mkv
    )
    shopt -u nullglob nocaseglob

    if [ "${#VIDEOS[@]}" -eq 0 ]; then
        echo "ERROR: no video files found under ${VIDEO_DIR}"
        exit 1
    fi

    echo "Select a video:"
    select selected in "${VIDEOS[@]}"; do
        if [ -n "${selected:-}" ]; then
            VIDEO_PATH="${selected}"
            break
        fi
        echo "Invalid selection. Enter one of the numbers above."
    done
fi

if [ ! -f "${VIDEO_PATH}" ]; then
    echo "ERROR: input video not found: ${VIDEO_PATH}"
    echo "Usage: bash test.sh [video filename or full path] [output_dir]"
    exit 1
fi

VIDEO_NAME="$(basename "${VIDEO_PATH}")"
VIDEO_STEM="${VIDEO_NAME%.*}"
OUTPUT_DIR="${2:-${OUTPUT_DIR:-${REPO_ROOT}/outputs/${VIDEO_STEM}_test}}"

# shellcheck disable=SC1091
source "${VENV_DIR}/bin/activate"
mkdir -p "${OUTPUT_DIR}"

echo "=============================================================="
echo "basketball-v2 quick video test"
echo "=============================================================="
echo "repo:       ${REPO_ROOT}"
echo "video:      ${VIDEO_PATH}"
echo "frames:     full video"
echo "tracker:    ${TRACKER}"
echo "output:     ${OUTPUT_DIR}"
echo "=============================================================="

cd "${REPO_ROOT}"
python src/pipeline.py \
  --input "${VIDEO_PATH}" \
  --config configs/default.yaml \
  --output-dir "${OUTPUT_DIR}" \
  --tracker "${TRACKER}"

echo ""
echo "Test complete. Outputs:"
ls -lh "${OUTPUT_DIR}"
