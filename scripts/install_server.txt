#!/usr/bin/env bash
#
# One-shot environment setup for a Linux GPU server.
#
#   cd /home/s1143757/basketball-v2
#   bash install_server.sh
#
# Direct invocation as `bash scripts/install_server.sh` is also supported.
#
# Order matters and is not negotiable:
#
#   1. torch, ALONE, from the PyTorch index matching this machine's CUDA driver
#   2. everything else from requirements.txt
#
# Installing torch as part of requirements.txt pulls it from plain PyPI, which
# can silently give you a CPU-only or mismatched-CUDA wheel. That build imports
# fine, trains fine, and runs roughly 40x slower - with no error to explain why.
# So this script reads the driver's CUDA version from nvidia-smi and picks the
# index URL, rather than guessing or hardcoding one.
#
# Safe to re-run. Nothing is installed outside the .venv it creates, and no
# sudo is required (see the tesseract section for the one optional exception).

set -u

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"

# Support both repository layouts:
#   /home/s1143757/basketball-v2/install_server.sh
#   /home/s1143757/basketball-v2/scripts/install_server.sh
# BASKETBALL_V2_ROOT can be set explicitly if the script is stored elsewhere.
if [ -n "${BASKETBALL_V2_ROOT:-}" ]; then
    REPO_ROOT="$(cd "${BASKETBALL_V2_ROOT}" && pwd)"
elif [ -e "${SCRIPT_DIR}/src/detect.py" ] && [ -e "${SCRIPT_DIR}/train/basketball.yaml" ]; then
    REPO_ROOT="${SCRIPT_DIR}"
elif [ -e "${SCRIPT_DIR}/../src/detect.py" ] && [ -e "${SCRIPT_DIR}/../train/basketball.yaml" ]; then
    REPO_ROOT="$(cd "${SCRIPT_DIR}/.." && pwd)"
else
    REPO_ROOT="/home/s1143757/basketball-v2"
fi
VENV_DIR="${REPO_ROOT}/.venv"
MIN_ULTRALYTICS="8.4.63"

PASS="[ OK ]"
FAIL="[FAIL]"
WARN="[WARN]"
STEP="====>"

fail() { echo ""; echo "${FAIL} $1"; echo ""; exit 1; }

echo "=============================================================="
echo "basketball-v2 server install"
echo "=============================================================="
echo "repo: ${REPO_ROOT}"
echo ""

# --------------------------------------------------------------------------
echo "${STEP} 1/7  Checking we are in the right place"
# --------------------------------------------------------------------------
for marker in "requirements.txt" "src/detect.py" "train/basketball.yaml"; do
    [ -e "${REPO_ROOT}/${marker}" ] || fail \
        "${REPO_ROOT}/${marker} not found. Run this from inside the basketball-v2 repo."
done
echo "${PASS} repo layout looks right"

# --------------------------------------------------------------------------
echo ""
echo "${STEP} 2/7  Locating a Python interpreter"
# --------------------------------------------------------------------------
PYTHON=""
for candidate in python3.12 python3.11 python3.10 python3 python; do
    if command -v "${candidate}" >/dev/null 2>&1; then
        version="$("${candidate}" -c 'import sys; print("%d.%d" % sys.version_info[:2])' 2>/dev/null)"
        major="${version%%.*}"
        minor="${version##*.}"
        if [ "${major}" = "3" ] && [ "${minor}" -ge 9 ] 2>/dev/null; then
            PYTHON="${candidate}"
            break
        fi
    fi
done
[ -n "${PYTHON}" ] || fail \
    "No Python 3.9+ found. On a cluster you usually need to load it first, e.g.
         module avail python
         module load python/3.11"
echo "${PASS} using ${PYTHON} ($(${PYTHON} --version 2>&1))"

# --------------------------------------------------------------------------
echo ""
echo "${STEP} 3/7  Creating the virtual environment"
# --------------------------------------------------------------------------
if [ -d "${VENV_DIR}" ]; then
    echo "${PASS} reusing existing ${VENV_DIR}"
else
    "${PYTHON}" -m venv "${VENV_DIR}" || fail \
        "venv creation failed - usually a missing python3-venv package.
         With no sudo, use conda instead and install by hand:
             conda create -n basketball python=3.11 -y
             conda activate basketball
             pip install torch torchvision --index-url https://download.pytorch.org/whl/cu121
             pip install -r ${REPO_ROOT}/requirements.txt
             python ${REPO_ROOT}/scripts/check_env.py
         (pick the cuXXX index URL matching 'CUDA Version' in nvidia-smi)"
    echo "${PASS} created ${VENV_DIR}"
fi

# shellcheck disable=SC1091
source "${VENV_DIR}/bin/activate" || fail "could not activate ${VENV_DIR}"
PIP="python -m pip"

# torch is ~2.5 GB. Home directories on shared clusters are often quota'd, and a
# pip cache that size is the usual way to hit it, so caching is off throughout.
export PIP_NO_CACHE_DIR=1

${PIP} install --upgrade pip setuptools wheel >/dev/null 2>&1 \
    && echo "${PASS} pip upgraded" \
    || echo "${WARN} pip upgrade failed; continuing"

# --------------------------------------------------------------------------
echo ""
echo "${STEP} 4/7  Detecting CUDA and installing torch"
# --------------------------------------------------------------------------
TORCH_INDEX=""
CUDA_LABEL=""

if command -v nvidia-smi >/dev/null 2>&1; then
    # nvidia-smi reports the highest CUDA version the DRIVER supports. Any torch
    # build at or below it works, so we step down to the nearest available one.
    DRIVER_CUDA="$(nvidia-smi 2>/dev/null | grep -o 'CUDA Version: *[0-9]*\.[0-9]*' | grep -o '[0-9]*\.[0-9]*' | head -1)"
    if [ -n "${DRIVER_CUDA}" ]; then
        echo "       driver supports CUDA up to ${DRIVER_CUDA}"
        CUDA_MAJOR="${DRIVER_CUDA%%.*}"
        CUDA_MINOR="${DRIVER_CUDA##*.}"
        if [ "${CUDA_MAJOR}" -ge 13 ] 2>/dev/null; then
            TORCH_INDEX="https://download.pytorch.org/whl/cu128"; CUDA_LABEL="cu128"
        elif [ "${CUDA_MAJOR}" = "12" ]; then
            if   [ "${CUDA_MINOR}" -ge 8 ] 2>/dev/null; then TORCH_INDEX="https://download.pytorch.org/whl/cu128"; CUDA_LABEL="cu128"
            elif [ "${CUDA_MINOR}" -ge 6 ] 2>/dev/null; then TORCH_INDEX="https://download.pytorch.org/whl/cu126"; CUDA_LABEL="cu126"
            elif [ "${CUDA_MINOR}" -ge 4 ] 2>/dev/null; then TORCH_INDEX="https://download.pytorch.org/whl/cu124"; CUDA_LABEL="cu124"
            else TORCH_INDEX="https://download.pytorch.org/whl/cu121"; CUDA_LABEL="cu121"; fi
        elif [ "${CUDA_MAJOR}" = "11" ]; then
            TORCH_INDEX="https://download.pytorch.org/whl/cu118"; CUDA_LABEL="cu118"
        fi
    fi
    nvidia-smi --query-gpu=name,memory.total --format=csv,noheader 2>/dev/null \
        | sed 's/^/       GPU: /'
else
    echo "${WARN} nvidia-smi not found."
    echo "       On a cluster the GPU is usually only visible inside a job, not on"
    echo "       the login node. If that is the case here, this is expected - but"
    echo "       you must then run this script inside an interactive GPU job, e.g."
    echo "           srun --gres=gpu:1 --pty bash"
    echo "       otherwise you will get a CPU-only torch."
fi

if [ -z "${TORCH_INDEX}" ]; then
    echo ""
    echo "${WARN} Could not determine a CUDA version, so the GPU build cannot be"
    echo "       chosen automatically. Installing the CPU build."
    echo "       Inference will be roughly 40x slower and training impractical."
    echo "       To fix later, inside the venv:"
    echo "           pip uninstall -y torch torchvision"
    echo "           pip install torch torchvision --index-url https://download.pytorch.org/whl/cu121"
    TORCH_INDEX="https://download.pytorch.org/whl/cpu"
    CUDA_LABEL="cpu"
fi

if python -c "import torch" >/dev/null 2>&1; then
    echo "${PASS} torch already installed: $(python -c 'import torch; print(torch.__version__)')"
else
    echo "       installing torch from ${TORCH_INDEX} (${CUDA_LABEL}) - this is a large download"
    ${PIP} install torch torchvision --index-url "${TORCH_INDEX}" \
        || fail "torch install failed. Check outbound network access, then retry:
             source ${VENV_DIR}/bin/activate
             pip install torch torchvision --index-url ${TORCH_INDEX}"
    echo "${PASS} torch installed"
fi

# --------------------------------------------------------------------------
echo ""
echo "${STEP} 5/7  Verifying torch actually sees the GPU"
# --------------------------------------------------------------------------
python - <<'PYEOF'
import sys
try:
    import torch
except Exception as exc:
    print(f"[FAIL] cannot import torch: {exc}")
    sys.exit(1)
print(f"[ OK ] torch {torch.__version__}")
if torch.cuda.is_available():
    print(f"[ OK ] CUDA available, built against CUDA {torch.version.cuda}")
    for i in range(torch.cuda.device_count()):
        props = torch.cuda.get_device_properties(i)
        print(f"[ OK ] device {i}: {props.name}, {props.total_memory / (1024**3):.1f} GB")
else:
    print("[WARN] torch.cuda.is_available() is False - this build will run on CPU.")
    print("       If you are on a login node without a GPU this is expected;")
    print("       re-check inside a GPU job. If you ARE on a GPU node, the wheel")
    print("       is CPU-only and must be reinstalled with the right index URL.")
PYEOF

# --------------------------------------------------------------------------
echo ""
echo "${STEP} 6/7  Installing the remaining dependencies"
# --------------------------------------------------------------------------
${PIP} install -r "${REPO_ROOT}/requirements.txt" \
    || fail "requirements.txt install failed. Re-run:
         source ${VENV_DIR}/bin/activate
         pip install -r requirements.txt"
echo "${PASS} requirements.txt installed"

python - "${MIN_ULTRALYTICS}" <<'PYEOF'
import sys

required = tuple(int(part) for part in sys.argv[1].split("."))
try:
    import ultralytics
except Exception as exc:
    print(f"[FAIL] cannot import ultralytics: {exc}")
    sys.exit(1)

version = getattr(ultralytics, "__version__", "0.0.0")
parts = []
for chunk in str(version).split(".")[:3]:
    digits = "".join(c for c in chunk if c.isdigit())
    parts.append(int(digits) if digits else 0)
while len(parts) < 3:
    parts.append(0)

if tuple(parts) < required:
    print(f"[FAIL] ultralytics {version} is older than the required {sys.argv[1]}.")
    print("       tracktrack.yaml is this project's DEFAULT tracker and was added")
    print("       in that release, so the default config cannot run at all.")
    print(f"       fix: pip install -U 'ultralytics>={sys.argv[1]}'")
    sys.exit(1)
print(f"[ OK ] ultralytics {version}")
PYEOF
[ $? -eq 0 ] || fail "ultralytics version check failed - see above"

# --------------------------------------------------------------------------
echo ""
echo "${STEP} 7/7  Checking the tesseract binary (optional)"
# --------------------------------------------------------------------------
# pytesseract is only a binding. The binary is a SYSTEM package that pip cannot
# install, and on a shared cluster you will usually not have sudo. This is
# deliberately non-fatal: without it the pipeline still produces all three
# outputs, and jersey numbers simply stay unknown while output 3 falls back to
# track ids. Nothing else degrades.
if command -v tesseract >/dev/null 2>&1; then
    echo "${PASS} tesseract found: $(tesseract --version 2>&1 | head -1)"
else
    echo "${WARN} tesseract binary not found on PATH."
    echo "       Consequence: jersey numbers stay unknown; output 3 falls back to"
    echo "       track ids. Everything else - shot detection, tracking, both"
    echo "       videos - works normally. Install it only if you want numbers:"
    echo "           with sudo:    sudo apt-get install -y tesseract-ocr"
    echo "           with conda:   conda install -c conda-forge tesseract"
    echo "           on a cluster: module avail tesseract"
fi

# --------------------------------------------------------------------------
echo ""
echo "=============================================================="
echo "Running the project's own environment check"
echo "=============================================================="
python "${REPO_ROOT}/scripts/check_env.py" --skip-download
CHECK_STATUS=$?

echo ""
echo "=============================================================="
if [ ${CHECK_STATUS} -eq 0 ]; then
    echo "Install complete."
else
    echo "Install finished, but check_env.py reported failures above."
fi
echo "=============================================================="
echo ""
echo "IMPORTANT - activate the venv in every new shell:"
echo "    source ${VENV_DIR}/bin/activate"
echo ""
echo "Next steps, in order:"
echo ""
echo "  1. Build the dataset (train/datasets/ is gitignored, so it is not"
echo "     in your upload and must be regenerated here):"
echo "         python scripts/prepare_dataset.py"
echo ""
echo "  2. Train, using the GENERATED yaml - not train/basketball.yaml,"
echo "     whose relative path resolves against Ultralytics' global"
echo "     DATASETS_DIR setting rather than this repo:"
echo "         python train/train.py train --data train/datasets/basketball/data.yaml"
echo ""
echo "  3. Only after weights exist, point detector.weights in"
echo "     configs/default.yaml at the produced best.pt and run:"
echo "         python src/pipeline.py --input clip.mp4 --output-dir outputs/run1"
echo ""
echo "The logic tests need no GPU and no weights - run them any time:"
echo "    python tests/test_jersey_logic.py"
echo "    python tests/test_shot_logic.py"
echo "    python tests/test_referee_exclusion.py"
echo ""
